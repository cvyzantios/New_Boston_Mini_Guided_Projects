        public string HairColor = "Brown";

        {

        }

        }
    }

// Delegates: μεταβλητές που μπορούν να αναφέρονται σε methods με συμβατή υπογραφή.

// ---------------------------------------------------------------------------

// Class: ορίζει έναν νέο τύπο/αντικείμενο στην C#.
// Class: defines a new type/object structure in C#.
   class MyClass
    {

        //Delegate pass the same parametres the same time
        delegate void MyDelegate(string myString); //Declare delegate
// Method: εκτελεί μια συγκεκριμένη λειτουργία και μπορεί να δέχεται parameters.
// Method: performs a specific operation and may receive parameters.
        public void ShowThoseMessages()
        {
// new: δημιουργεί νέο instance/object του συγκεκριμένου type.
// new: creates a new instance/object of the specified type.
            MyDelegate md = new MyDelegate(ShowMessage);
            md += ShowAnotherMessage;
            md("Adam");
        }
// Method: εκτελεί μια συγκεκριμένη λειτουργία και μπορεί να δέχεται parameters.
// Method: performs a specific operation and may receive parameters.
        void ShowAnotherMessage(string message)
        {
// MessageBox.Show: εμφανίζει μήνυμα σε παράθυρο WinForms.
// MessageBox.Show: displays a message in a WinForms dialog.
            System.Windows.Forms.MessageBox.Show(message);
        }
